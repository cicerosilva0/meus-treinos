Como conectar ao Google Sheets

Para salvar os dados dos usuários na sua planilha, você precisa criar um Google Apps Script que funciona como endpoint. Siga os passos abaixo:

1
Abra o Google Sheets — acesse sua planilha e vá em Extensões → Apps Script
2
Cole o seguinte código no editor e salve (Ctrl+S):
function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName("Usuarios") || SpreadsheetApp.getActiveSpreadsheet() .insertSheet("Usuarios"); // Cabeçalho se vazio if (sheet.getLastRow() === 0) { sheet.appendRow([ "Nome","E-mail","Cadastro","Treinos Totais", "Semanas","Volume (kg)","Sequência","Última Sincronização" ]); sheet.getRange(1,1,1,8).setFontWeight("bold") .setBackground("#c8f25e").setFontColor("#111111"); } var users = data.users || []; // Limpar dados antigos (manter header) if (sheet.getLastRow() > 1) { sheet.getRange(2, 1, sheet.getLastRow()-1, 8).clearContent(); } users.forEach(function(u) { sheet.appendRow([ u.nome, u.email, u.cadastro, u.treinos, u.semanas, u.volume, u.streak, new Date().toLocaleString("pt-BR") ]); }); return ContentService .createTextOutput(JSON.stringify({ok: true, count: users.length})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ok: false, error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet() .getSheetByName("Usuarios"); if (!sheet || sheet.getLastRow() < 2) { return ContentService .createTextOutput(JSON.stringify({ok: true, users: []})) .setMimeType(ContentService.MimeType.JSON); } var rows = sheet.getRange(2, 1, sheet.getLastRow()-1, 7).getValues(); var users = rows.map(function(r) { return {nome:r[0],email:r[1],cadastro:r[2],treinos:r[3],semanas:r[4],volume:r[5],streak:r[6]}; }); return ContentService .createTextOutput(JSON.stringify({ok: true, users: users})) .setMimeType(ContentService.MimeType.JSON); }
3
Clique em Implantar → Gerenciar implantações. Se já implantou antes, clique no ✏️ (editar) → em "Versão" escolha "Nova versão" → Implantar. Isso aplica o código novo.

Se for a primeira vez: Implantar → Nova implantação → tipo Aplicativo da Web → "Executar como" = Eu → "Quem tem acesso" = Qualquer pessoa → Implantar.
